Each page presents one case study as a slide viewer. All slide images are
pre-extracted JPEGs in a `slides/` folder sitting next to the HTML files.

## Lightbox

```html
<div id="lightbox" class="lightbox">
  <button class="lb-close" onclick="closeLightbox()">✕</button>
  <div class="lb-img-wrap">
    <img id="lb-img" src="" alt="">
  </div>
  <div class="lb-info">
    <h3 id="lb-title"></h3>
    <p id="lb-desc"></p>
  </div>
  <div id="lb-counter" class="lb-counter"></div>
  <div class="lb-nav">
    <button onclick="lbNav(-1)">← Prev</button>
    <button onclick="lbNav(1)">Next →</button>
  </div>
</div>
```

**Styles:**
- `position:fixed; inset:0; display:none` → `display:flex` when `.open`
- `background: rgba(10,10,8,0.92)`, `backdrop-filter: blur(6px)`
- Image: `max-width: 92vw; max-height: 78vh; border-radius: 8px`
- `.lb-close`: top-right circle button, white

**JS:**
```js
// Build this array from slide metadata — file paths + titles + descs
const slides = [
  { src: "slides/cs1-03.jpg", title: "Packaging — Series of Three", desc: "Title Slide" },
  // ... all slides for this page
];
let current = 0;

function openLightbox(i) {
  current = i;
  updateLightbox();
  document.getElementById('lightbox').classList.add('open');
  document.body.style.overflow = 'hidden';
}
function closeLightbox() {
  document.getElementById('lightbox').classList.remove('open');
  document.body.style.overflow = '';
}
function updateLightbox() {
  const s = slides[current];
  document.getElementById('lb-img').src = s.src;
  document.getElementById('lb-title').textContent = s.title;
  document.getElementById('lb-desc').textContent = s.desc;
  document.getElementById('lb-counter').textContent = (current+1) + ' / ' + slides.length;
}
function lbNav(dir) {
  current = (current + dir + slides.length) % slides.length;
  updateLightbox();
}
document.getElementById('lightbox').addEventListener('click', e => {
  if (e.target === document.getElementById('lightbox')) closeLightbox();
});
document.addEventListener('keydown', e => {
  if (!document.getElementById('lightbox').classList.contains('open')) return;
  if (e.key === 'ArrowRight') lbNav(1);
  if (e.key === 'ArrowLeft') lbNav(-1);
  if (e.key === 'Escape') closeLightbox();
});
```

## Important Notes
- Images are referenced as `slides/FILENAME.jpg` — relative paths, no base64
- Use `loading="lazy"` on every slide img tag
- No frameworks, no bundlers — vanilla HTML/CSS/JS only
- Each HTML file is otherwise fully self-contained (inline CSS + JS)
- The nav links between pages: case-study-1.html / case-study-2.html / case-study-3.html
- Deploy as a folder: the 3 HTML files + the slides/ subfolder
